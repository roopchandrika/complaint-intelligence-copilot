-- P2. Business question (API lookup): What were Wells Fargo's top issues in 2025?
SELECT issue, count(*) AS n
FROM complaints
WHERE company = 'WELLS FARGO & COMPANY'
  AND date_received >= DATE '2025-01-01' AND date_received < DATE '2026-01-01'
GROUP BY issue
ORDER BY n DESC
